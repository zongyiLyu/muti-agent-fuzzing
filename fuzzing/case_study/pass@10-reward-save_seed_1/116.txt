    def count_ones(num):
        return bin(num).count('1')

    positive_nums = [num for num in arr if num >= 0]
    negative_nums = [abs(num) for num in arr if num < 0]

    positive_nums.sort(key=lambda x: (count_ones(x), x))
    negative_nums.sort()

    negative_nums = [-num for num in negative_nums]

    return negative_nums + positive_nums

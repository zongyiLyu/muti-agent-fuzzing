    You are provided with a string that indicates the count of apples and oranges in a basket, along with an integer representing the total number of fruits. Your task is to determine the number of mangoes.
    For example:
    fruit_distribution("5 apples and 6 oranges", 19) -> 19 - 5 - 6 = 8
    fruit_distribution("0 apples and 1 oranges", 3) -> 3 - 0 - 1 = 2

    """

def fruit_distribution(s, n):
    def extract_count(s):
        counts = {'apples': 0, 'oranges': 0}
        for item in s.split():
            if item.isdigit():
                counts['apples'] = int(item)
            elif item == 'apples':
                counts['apples'] = int(s.split()[s.split().index(item) - 1])

        return counts

    total_fruits = counts['apples'] + counts['oranges']
    mangoes = n - total_fruits
    return mangoes

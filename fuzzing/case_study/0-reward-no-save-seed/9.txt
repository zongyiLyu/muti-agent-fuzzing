from typing import List, Tuple

def rolling_max(numbers: List[int]) -> List[int]:
    if not numbers:
        return []
    
    k = 3  # Size of the initial subarray
    rolling_max_list = []
    max_element = max(numbers[:k])
    rolling_max_list.append(max_element)
    
    for i in range(k, len(numbers)):
        if numbers[i] >= max_element:
            max_element = numbers[i]
        rolling_max_list.append(max_element)
    
    return rolling_max_list

from typing import List

def rolling_max(numbers: List[int]) -> List[int]:
    if not numbers:
        return []

    k = 3  # Size of the sliding window
    rolling_max_list = []
    window = []

    for i, num in enumerate(numbers):
        if i >= k and window[0] <= i - k:
            window.pop(0)

        while window and numbers[window[-1]] <= num:
            window.pop()

        window.append(i)

        if i >= k - 1:
            rolling_max_list.append(numbers[window[0]])

    return rolling_max_list

    if len(lst1) != len(lst2):
        return "NO"
    
    if (sum(lst1) + sum(lst2)) % 2 != 0:
        return "NO"
    
    count_even_lst1 = sum(1 for num in lst1 if num % 2 == 0)
    count_even_lst2 = sum(1 for num in lst2 if num % 2 == 0)
    
    if (count_even_lst1 - count_even_lst2) % 2 != 0:
        return "NO"
    
    return "YES"

    A string S is considered to be good if and only if all parentheses in S

    Examples:
    match_parens(['()(', ')']) == 'Yes'
    match_parens([')', ')']) == 'No'
    '''

def match_parens(lst):
    if len(lst) != 2:
        return 'No'
    
    s1, s2 = lst[0], lst[1]
    
    if any(char not in '()' for char in s1 + s2):
        return 'No'
    
    possible_orders = [s1 + s2, s2 + s1]
    
    for order in possible_orders:
        if is_balanced(order):
            return 'Yes'
    
    return 'No'

from typing import List

def factorize(n: int) -> List[int]:
    def prime_factors(num):
        factors = []
        divisor = 2
        while divisor <= num:
            if num % divisor == 0:
                factors.append(divisor)
                num = num / divisor
            else:
                divisor += 1
        return factors

    def count_occurrences(factors):
        occurrences = {}
        for factor in factors:
            if factor in occurrences:
                occurrences[factor] += 1
            else:
                occurrences[factor] = 1
        return occurrences

    def generate_sorted_factors(occurrences):
        sorted_factors = []
        for factor, count in sorted(occurrences.items(), key=lambda x: x[1]):
            sorted_factors.extend([factor] * count)
        return sorted_factors

    factors = prime_factors(n)
    occurrences = count_occurrences(factors)
    sorted_factors = generate_sorted_factors(occurrences)

    return sorted_factors

import math

def factorize(n: int) -> List[int]:
    factors = []

    if num < 2:
        return False
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            return False
    return True

    factors = []
    i = 2
    while i <= n:
        if n % i == 0 and is_prime(i):
            factors.append(i)
            n = n // i
        else:
            i += 1
    return factors

        if factor in occurrences:
            occurrences[factor] += 1
        else:
            occurrences[factor] = 1
    return occurrences

    prime_factors = find_prime_factors(n)

    result = []

    return result

from typing import List

def is_prime(num: int) -> bool:
    if num < 2:
        return False
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            return False
    return True

def find_prime_factors(num: int) -> List[int]:
    factors = []
    divisor = 2
    while divisor <= num:
        if num % divisor == 0:
            factors.append(divisor)
            num = num // divisor
        else:
            divisor += 1
    return factors

def arrange_factors(factors: List[int]) -> List[int]:
    return sorted(factors)

def factorize(n: int) -> List[int]:
    if n == 1:
        return [1]
    if is_prime(n):
        return [n]
    prime_factors = find_prime_factors(n)
    arranged_factors = arrange_factors(prime_factors)
    return arranged_factors

    A string S is considered to be good if and only if all parentheses in S

    Examples:
    match_parens(['()(', ')']) == 'Yes'
    match_parens([')', ')']) == 'No'
    '''

def match_parens(lst):
    open_count_1 = lst[0].count('(')
    close_count_1 = lst[0].count(')')
    open_count_2 = lst[1].count('(')
    close_count_2 = lst[1].count(')')

def match_parens(lst):
    open_count_1 = lst[0].count('(')
    close_count_1 = lst[0].count(')')
    open_count_2 = lst[1].count('(')
    close_count_2 = lst[1].count(')')

    if open_count_1 != close_count_1 or open_count_2 != close_count_2:
        return 'No'

    concatenated = lst[0] + lst[1]
    n = len(concatenated)
    for i in range(1 << n):
        candidate = ''.join([concatenated[j] for j in range(n) if (i & (1 << j))])
        if is_balanced(candidate):
            return 'Yes'
    
    return 'No'

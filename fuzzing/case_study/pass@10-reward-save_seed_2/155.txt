        raise ValueError("Input must be a valid integer")

    odd_count = 0

        if digit % 2 == 0:
            even_count += 1
        else:

    num_str = str(abs(num))
    even_count = sum(1 for digit in num_str if int(digit) % 2 == 0)
    odd_count = len(num_str) - even_count
    if num < 0:
        return (odd_count, even_count)
    return (even_count, odd_count)

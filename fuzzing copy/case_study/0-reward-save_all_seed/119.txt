def match_parens(lst):
    # Verify that the input list contains exactly two strings
    if len(lst) != 2 or not all(isinstance(s, str) for s in lst):
        return 'No'
    
    # Verify that both strings consist of only open or close parentheses
    for s in lst:
        if not all(c in ['(', ')'] for c in s):
            return 'No'
    
    # Concatenate the two strings in all possible orders
    concat_options = [lst[0] + lst[1], lst[1] + lst[0]]
    
    # Check if the resulting string is good by verifying if all parentheses are balanced
    def is_balanced(s):
        stack = []
        for char in s:
            if char == '(':
                stack.append(char)
            elif char == ')':
                if not stack:
                    return False
                stack.pop()
        return len(stack) == 0
    
    for option in concat_options:
        if is_balanced(option):
            return 'Yes'
    
    return 'No'

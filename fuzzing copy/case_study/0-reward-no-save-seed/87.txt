def get_row(lst, x):
    result = []
    for i, row in enumerate(lst):
        for j, val in enumerate(row):
            if val == x:
                result.append((i, j))
    
    result.sort(key=lambda tup: (tup[0], -tup[1]))
    
    return result

def get_row(lst, x):
    def sort_columns_desc(row):
        return sorted(enumerate(row), key=lambda x: x[1], reverse=True)
    
    rows_with_x = [(i, row) for i, row in enumerate(lst) if x in row]
    sorted_rows = sorted(rows_with_x, key=lambda x: x[0])
    
    result = []
    for row_idx, row in sorted_rows:
        sorted_columns = sort_columns_desc(row)
        for col_idx, _ in sorted_columns:
            result.append((row_idx, col_idx))
    
    return result

def get_row(lst, x):
    def extract_rows(lst, x):
        return [(i, row) for i, row in enumerate(lst) if x in row]

    def sort_rows(rows):
        return sorted(rows, key=lambda x: x[0])

    def sort_columns(row):

    extracted_rows = extract_rows(lst, x)
    sorted_rows = sort_rows(extracted_rows)
    result = [(i, j) for i, row in sorted_rows for j in sort_columns((i, row))]
    
    return result

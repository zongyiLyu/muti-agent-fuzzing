    """Perform a circular shift on the digits of the integer x, moving the digits to the right by the specified shift amount, and return the outcome as a string.
    If the shift value exceeds the number of digits, return the digits in reverse order.
    >>> circular_shift(12, 1)
    "21"
    >>> circular_shift(12, 2)
    "12"
    """

def circular_shift(x, shift):
    x_str = str(x)
    num_digits = len(x_str)
    actual_shift = shift % num_digits
    if actual_shift == 0:
        return x_str
    else:
        shifted_str = x_str[-actual_shift:] + x_str[:-actual_shift]
        if actual_shift > num_digits:
            return x_str[::-1]
        else:
            return shifted_str

def solve(s):
    """For a given string s, change the case of each alphabetic character. 
    If the string contains no alphabetic characters, simply reverse the string.
    Return the modified string.
    You should handle both uppercase and lowercase alphabetic characters appropriately.
    Examples
    solve("1234") = "4321"
    solve("ab") = "AB"
    solve("#a@C") = "#A@c"
    """

    flg = 0
    idx = 0
    new_str = list(s)
    for i in s:
        if i.isalpha():
            new_str[idx] = i.swapcase()
            flg = 1
        idx += 1
    s = ""
    for i in new_str:
        s += i
    if flg == 0:
        return s[len(s)::-1]
    return s

def solve(s):
    result = ''
    has_letter = False
    for char in s:
        if char.isalpha():
            has_letter = True
            if char.islower():
                result += char.upper()
            else:
                result += char.lower()
        else:
            result += char
    if not has_letter:
        result = result[::-1]
    return result

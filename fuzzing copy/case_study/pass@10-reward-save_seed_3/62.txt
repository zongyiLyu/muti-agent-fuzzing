def derivative(xs: list):
    """ xs represent coefficients of a polynomial.
    xs[0] + xs[1] * x + xs[2] * x^2 + ....
    Return derivative of this polynomial in the same form.
    >>> derivative([3, 1, 2, 4, 5])
    [1, 4, 12, 20]
    >>> derivative([1, 2, 3])
    [2, 6]
    The input list will always have at least one element.
    The coefficients are given in increasing order of powers of x.
    The derivative of a constant term is zero and should be omitted.
    """
------------------------------
canonical solution
------------------------------
    return [(i * x) for i, x in enumerate(xs)][1:]

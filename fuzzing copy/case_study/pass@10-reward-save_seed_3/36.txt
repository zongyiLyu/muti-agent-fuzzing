    return str(num).count('7')

def is_divisible_by_11_or_13(num):
    return num % 11 == 0 or num % 13 == 0

def generate_integers_less_than_n(n):
    return [i for i in range(n)]

    integers = generate_integers_less_than_n(n)
    filtered_integers = [num for num in integers if count_sevens(num) > 0 and is_divisible_by_11_or_13(num)]
    return len(filtered_integers)

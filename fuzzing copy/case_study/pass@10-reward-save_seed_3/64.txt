def vowels_count(s):
    vowels = set(['a', 'e', 'i', 'o', 'u'])
    count = 0
    for i in range(len(s)):
        if s[i].lower() in vowels:
            if s[i].lower() == 'y' and i != len(s) - 1:
                continue
            count += 1
    return count

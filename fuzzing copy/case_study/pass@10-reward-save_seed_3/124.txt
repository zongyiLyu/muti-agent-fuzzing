def valid_date(date):
    if not date:
        return False
    
    try:
        month, day, year = map(int, date.split('-'))
    except ValueError:
        return False
    
    if month < 1 or month > 12:
        return False
    
    if month in [1, 3, 5, 7, 8, 10, 12]:
        max_days = 31
    elif month in [4, 6, 9, 11]:
        max_days = 30
    else:
        if year % 4 == 0 and (year % 100 != 0 or year % 400 == 0):
            max_days = 29
        else:
            max_days = 28
    
    if day < 1 or day > max_days:
        return False
    
    return True

        return False

        return False
    
    month = int(month)
    day = int(day)
    year = int(year)

        if day < 1 or day > 31:

            return False
    elif month == 2:
        if year % 4 == 0:

                return False
        else:
            if day < 1 or day > 28:
                return False
    
    return True

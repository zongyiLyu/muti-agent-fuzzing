        search([5, 5, 4, 4, 4]) == -1
    '''

def search(lst):
    '''
    You have a non-empty list of positive integers. Your task is to find and return the largest integer that is greater than zero and appears in the list with a frequency that is at least equal to the integer's value. 
    The frequency of an integer refers to how many times it shows up in the list.
    If no such integer is found, return -1.
    Examples:
        search([4, 1, 2, 2, 3, 1]) == 2
        search([1, 2, 2, 3, 3, 3, 4, 4, 4]) == 3
        search([5, 5, 4, 4, 4]) == -1
    '''

    frq = [0] * (max(lst) + 1)
    for i in lst:
        frq[i] += 1;

    ans = -1
    for i in range(1, len(frq)):
        if frq[i] >= i:
            ans = i
    
    return ans

def search(lst):
    freq_dict = {}
    
    for num in lst:
        if num in freq_dict:
            freq_dict[num] += 1
        else:
            freq_dict[num] = 1
    
    result = -1
    for key, value in freq_dict.items():
        if key > 0 and key <= value:
            result = max(result, key)
    
    return result

def odd_count(lst):
    def count_odd_digits(s):
        return sum(1 for digit in s if int(digit) % 2 != 0)

    output = []
    for string in lst:
        odd_count = count_odd_digits(string)
        output.append(f"the number of odd elements {odd_count}n the str{odd_count}ng {odd_count} of the {odd_count}nput.")

    return output

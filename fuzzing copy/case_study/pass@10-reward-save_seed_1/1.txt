from typing import List

def separate_paren_groups(paren_string: str) -> List[str]:
    # Step 1: Remove spaces from the input string
    paren_string = paren_string.replace(" ", "")
    
    # Step 2: Loop through the input string and identify separate groups of nested parentheses
    groups = []
    current_group = ""
    open_braces = 0
    
    for char in paren_string:
        if char == "(":
            open_braces += 1
        elif char == ")":
            open_braces -= 1
        
        current_group += char
        
        if open_braces == 0:
            groups.append(current_group)
            current_group = ""
    
    # Step 3: Check if each group is balanced by counting the number of open and close parentheses
    for group in groups:
        if group.count("(") != group.count(")"):
            raise ValueError("Unbalanced parentheses in group: {}".format(group))
    
    # Step 4: Return the list of separated groups
    return groups

# Test the function
print(separate_paren_groups('( ) (( )) (( )( ))'))  # Output: ['()', '(())', '(()())']

from typing import List

def separate_paren_groups(paren_string: str) -> List[str]:
    def extract_groups(paren_string):
        groups = []
        group = ""
        count = 0
        for char in paren_string:
            if char == '(':
                count += 1
                if count == 1:
                    group += char
            elif char == ')':
                count -= 1
                if count == 0:
                    group += char
                    groups.append(group)
                    group = ""
            else:
                if count > 0:
                    group += char
        return groups

    def remove_spaces(groups):
        return [group.replace(" ", "") for group in groups]

    groups = extract_groups(paren_string)
    individual_strings = remove_spaces(groups)
    
    return individual_strings

    def calculate_collision_time(pos1, pos2):
        if pos1[0] < pos2[0] and pos1[1] > pos2[1]:
            return (pos2[0] - pos1[0]) / (pos1[1] - pos2[1])
        return float('inf')

    positions = [(0, 1) for _ in range(n)] + [(0, -1) for _ in range(n)]
    collisions = 0

    for i in range(2 * n):
        for j in range(i + 1, 2 * n):
            collision_time = calculate_collision_time(positions[i], positions[j])
            if collision_time < float('inf'):
                collisions += 1

    return collisions
